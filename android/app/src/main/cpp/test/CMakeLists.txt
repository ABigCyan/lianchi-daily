# 在电脑上测试内置模型核心（llm-core.cpp，和手机上同一份代码）：bash tools/test-llm-core.sh
cmake_minimum_required(VERSION 3.22.1)
project(llm_core_test C CXX)
set(LLAMA_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../llama.cpp)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
add_subdirectory(${LLAMA_DIR} llama)
add_executable(llm-core-test llm-core-test.cpp ../llm-core.cpp)
target_include_directories(llm-core-test PRIVATE ..)
target_compile_features(llm-core-test PRIVATE cxx_std_17)
target_link_libraries(llm-core-test PRIVATE llama mtmd)
